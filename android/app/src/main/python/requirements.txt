# 11.0.0 is the newest Pillow chaquopy ships a wheel for
# (https://chaquo.com/pypi-13.1/pillow/, checked 2026-09-15,
# PyPI has no android wheel at all). that puts it under every
# Pillow advisory fixed in 12.x. all but one live in codecs
# and font/ICC/PDF paths this app never calls: the only Pillow
# work here is frombytes on Unity texture data, crop, resize,
# alpha_composite and a PNG save. the one that does apply is
# CVE-2026-59199 (crop box overflow -> heap write), and
# recover_assets.py bounds every crop before Pillow sees it, so
# this stays safe as long as that check does. the dependabot
# alerts on the public repo are dismissed with this reasoning.
# bump the moment a 12.3+ arm64 wheel shows up.
Pillow==11.0.0
lz4==4.3.2
attrs==26.1.0
fsspec==2026.7.0

# pip runs with --no-deps (UnityPy's compiled dependencies and
# brotli have no usable Android wheels and are stubbed in this
# source set), so Pillow's native support libraries have to be
# requested by hand - without them PIL/_imaging.so has no
# libjpeg_chaquopy.so to dlopen.
chaquopy-libjpeg==1.5.3
chaquopy-freetype==2.9.1

# UnityPy itself is not installed here: unpackUnityPy in
# build.gradle.kts drops the pure-Python tree into the source set
# instead.
